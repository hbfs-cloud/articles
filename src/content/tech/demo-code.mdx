<ContentCard title="SQL — Point-in-Time Queries" id="sql-section">
  <CodeBlock label="SQL — SCD Type 2 Pattern">
    ```sql
    -- The instruments table stores every version of every security
    SELECT symbol, name, sector, exchange,
           valid_from, valid_to
    FROM   instruments
    WHERE  symbol = 'AAPL'
      AND  valid_from <= '2017-06-15'
      AND  (valid_to > '2017-06-15' OR valid_to IS NULL);

    -- Create a materialized view for fast lookups
    CREATE MATERIALIZED VIEW mv_current_instruments AS
    SELECT *
    FROM   instruments
    WHERE  valid_to IS NULL
    WITH DATA;

    -- Partition by exchange for performance
    CREATE TABLE prices (
        symbol     TEXT NOT NULL,
        trade_date DATE NOT NULL,
        open       NUMERIC(12,4),
        high       NUMERIC(12,4),
        low        NUMERIC(12,4),
        close      NUMERIC(12,4),
        volume     BIGINT
    ) PARTITION BY RANGE (trade_date);
    ```
  </CodeBlock>

  <PedagogyBox title="Why SCD Type 2?">
    Slowly Changing Dimensions Type 2 keeps a full history of every change.
    When Apple changed its sector from "Consumer Electronics" to "Technology",
    both versions are preserved with `valid_from` / `valid_to` timestamps.
  </PedagogyBox>
</ContentCard>

<ContentCard title="Python — Data Pipeline" id="python-section">
  <CodeBlock label="Python — Fetching market data with async">
    ```python
    import asyncio
    import aiohttp
    from dataclasses import dataclass
    from datetime import datetime

    @dataclass
    class Quote:
        symbol: str
        price: float
        volume: int
        timestamp: datetime

    async def fetch_quotes(symbols: list[str]) -> list[Quote]:
        """Fetch real-time quotes from MarketWatch Gateway."""
        async with aiohttp.ClientSession() as session:
            tasks = [_fetch_one(session, sym) for sym in symbols]
            return await asyncio.gather(*tasks)

    async def _fetch_one(session, symbol: str) -> Quote:
        url = f"https://api.market-watch.xyz/v1/quote/{symbol}"
        async with session.get(url) as resp:
            data = await resp.json()
            return Quote(
                symbol=data["symbol"],
                price=data["last"],
                volume=data["volume"],
                timestamp=datetime.fromisoformat(data["timestamp"]),
            )

    # Usage
    if __name__ == "__main__":
        quotes = asyncio.run(fetch_quotes(["AAPL", "NVDA", "MSFT"]))
        for q in quotes:
            print(f"{q.symbol}: ${q.price:.2f} ({q.volume:,} shares)")
    ```

<ContentCard title="Go — MCP Server" id="go-section">
  <CodeBlock label="Go — MCP Tool Handler">
    ```go
    package main

    import (
        "context"
        "encoding/json"
        "fmt"
        "log"

        "github.com/mark3labs/mcp-go/mcp"
        "github.com/mark3labs/mcp-go/server"
    )

    type QueryParams struct {
        Symbol string `json:"symbol"`
        From   string `json:"from"`
        To     string `json:"to"`
    }

    func handleQuery(ctx context.Context, req mcp.CallToolRequest) (*mcp.CallToolResult, error) {
        var params QueryParams
        raw, _ := json.Marshal(req.Params.Arguments)
        if err := json.Unmarshal(raw, &params); err != nil {
            return nil, fmt.Errorf("invalid params: %w", err)
        }

        rows, err := db.QueryContext(ctx,
            `SELECT trade_date, close, volume
             FROM   prices
             WHERE  symbol = $1
               AND  trade_date BETWEEN $2 AND $3
             ORDER BY trade_date`,
            params.Symbol, params.From, params.To,
        )
        if err != nil {
            return nil, err
        }
        defer rows.Close()

        var results []map[string]any
        for rows.Next() {
            var date string
            var close float64
            var volume int64
            rows.Scan(&date, &close, &volume)
            results = append(results, map[string]any{
                "date":   date,
                "close":  close,
                "volume": volume,
            })
        }

        data, _ := json.Marshal(results)
        return mcp.NewToolResultText(string(data)), nil
    }

    func main() {
        s := server.NewMCPServer("secmaster", "1.0.0")
        s.AddTool(mcp.NewTool("query_prices",
            mcp.WithDescription("Query historical prices"),
            mcp.WithString("symbol", mcp.Required()),
            mcp.WithString("from"),
            mcp.WithString("to"),
        ), handleQuery)

        log.Fatal(server.ServeStdio(s))
    }
    ```
